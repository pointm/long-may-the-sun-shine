% 定义一个宏命令来绘制题目部分
\newcommand{\drawQuestions}[2]{
    \section{#1}
    \begin{center}
    \begin{tikzpicture}
    % 定义局部变量
    \def\boxheight{-22}
    \def\boxwidth{16.8}
    \def\verticalspacing{4em}
    \def\horizontalspacing{.25\linewidth}
    \def\linelength{1.5cm}
    % 绘制固定的矩形框
    \draw[very thick] (0.21,0) rectangle (\boxwidth,\boxheight);
    % 使用嵌套循环
    \def\colomnumber{#2} % 定义每一列数字的个数
    \foreach \j in {1, 2, 3, 4} {
        \foreach \i in {1, 2, 3, ..., \colomnumber} {
            \pgfmathsetmacro{\result}{\i + \colomnumber*(\j-1)}
            \node at ({(\j-1)*\horizontalspacing + 0.5*\horizontalspacing},{\boxheight - \i*\verticalspacing + 0.5*\verticalspacing}){\pgfmathprintnumber[fixed, precision=0]{\result}.\, \underline{\hspace{\linelength}}};
        }
    }
    \end{tikzpicture}
    \end{center}
    \newpage
}
